    Column("role_id", UUID(as_uuid=True), ForeignKey("roles.id")),

)

class Role(Base):

"""Rate limiting implementation using Redis."""
from datetime import datetime
from fastapi import HTTPException, Request
from core.cache import redis_cache

class RateLimiter:
    """Rate limiter using Redis."""
    
    def __init__(self, requests_per_minute: int = 60):
        self.requests_per_minute = requests_per_minute
        self.cache = redis_cache
    
    async def check_rate_limit(self, request: Request, user_id: str = None):
        """Check rate limit for request."""
        now = datetime.utcnow().timestamp()
        key = f"ratelimit:{user_id or request.client.host}"
        
        # Use Redis pipeline for atomic operations
        pipe = self.cache.redis.pipeline()
        pipe.zremrangebyscore(key, 0, now - 60)  # Remove old requests
        pipe.zadd(key, {str(now): now})  # Add current request
        pipe.zcard(key)  # Count requests
        pipe.expire(key, 60)  # Set expiry
        _, _, requests, _ = pipe.execute()
        
        if requests > self.requests_per_minute:
            raise HTTPException(
                status_code=429,
                detail="Too many requests"
            )

async def check_permission(user: User, required_permission: str) -> bool:
    """Check if user has required permission."""
    user_permissions = set()
    for role in user.roles:
        user_permissions.update(perm.name for perm in role.permissions)
    return required_permission in user_permissions

def requires_permission(permission: str):
    """Dependency for checking permission."""
    async def permission_checker(
        current_user: User = Depends(get_current_active_user)
    ):
        if not await check_permission(current_user, permission):
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Not enough permissions"
            )
        return current_user
    return permission_checker
